@app.post("/exam-questions")
def exam_questions(req: ExamRequest, x_api_key: str = Header(default="")):
    verify_api_key(x_api_key)                                  # ① 인증

    segments = get_slide_segments(req.keyword)                 # ② 근거 조회
    if not segments:
        raise HTTPException(status_code=404, detail="슬라이드 내용을 찾을 수 없습니다.")
    slide_text = join_slide_text(segments)
    reference_questions = get_virtual_questions(req.keyword, req.level)

    try:
        questions = generate_exam_questions(                    # ③ 체인 호출
            slide_text, req.question_type, req.level, req.count, reference_questions
        )
    except RuntimeError as err:
        raise HTTPException(status_code=500, detail=str(err))

    return {"questions": [q.model_dump() for q in questions]}

1. LangChain 공통 진입점 — llm_provider.py

프로젝트 전체에서 ChatOpenAI를 생성하는 유일한 파일이다. 체인들은 모델명·키를 모르고 이 함수만 부른다.

# llm_provider.py:21-28
def get_openai_llm(temperature: float = 0, timeout: float | None = None) -> ChatOpenAI:
    if not OPENAI_KEY:
        raise RuntimeError("OPENAI_KEY 환경변수가 설정되어 있지 않습니다.")
    model_name = os.environ.get("OPENAI_MODEL", DEFAULT_MODEL)   # 기본 gpt-4o
    kwargs = {"model": model_name, "api_key": OPENAI_KEY, "temperature": temperature}
    if timeout is not None:
        kwargs["timeout"] = timeout
    return ChatOpenAI(**kwargs)

temperature=0 고정, 모델은 Cloud Run 환경변수 OPENAI_MODEL로 교체 가능 — 코드 수정 없이 모델을 바꾸기 위한 설계다.

bind_tools()로 OpenAI 호스티드 웹검색을 붙인다. 검색은 OpenAI 서버가 알아서 하므로 클라이언트가 ReAct 루프를 돌 필요가 없고, 호출부는 .invoke() 한 번뿐이다.

검색과 요약을 2단계로 나눈 이유 - 
검색 도구가 걸린 호출에 "한국어로", "마크다운 금지" 같은 스타일 지시를 같이 주면 모델이 "찾은 내용을 보고하는" 쪽으로 쏠려 지시를 안 지킨다. 
그래서 ① 검색 전용 호출로 원시 결과만 받고(언어·형식 무시), ② 도구 없는 일반 호출로 한국어 요약을 다시 쓴다.

5. LangGraph — 가상질문 생성 (virtual_question_agent.py)

유일한 LangGraph 사용처다. 다른 기능은 호출 1~2회로 끝나 상태 유지가 필요 없지만, 
이 기능은 슬라이드를 Part 1→2→3으로 순서대로 읽으며 앞 구간 맥락을 누적해야 해서 상태가 필요하다.

그래프 구조

START → read_batch ─┐
            ↑        │ (batch_index < 배치 개수: 다음 구간으로 루프)
            └────────┘
                     │ (마지막 구간까지 끝나면)
                     ↓
              filter_questions → END

상태 정의 (TypedDict)

# chains/virtual_question_agent.py:64-72
class _VqState(TypedDict):
    persona_prompt: str          # 페르소나 프롬프트(Supabase에서 조회)
    batches: list[dict]          # Part 1/2/3 분할 결과
    batch_index: int             # 현재 구간 - 조건부 엣지의 판정 기준
    use_summary: bool            # 50장 초과 여부 → 맥락 누적 방식 결정
    context_text: str            # 원문 누적(50장 이하)
    context_summaries: list[str] # 요약 누적(50장 초과)
    all_questions: list[dict]    # 구간별 원본 질문 누적
    final_questions: list[dict]  # 중복 정리 후 최종본(반환·저장 대상)

노드 1 — read_batch (루프의 몸통)

각 노드는 상태 전체가 아니라 바뀐 부분만 dict로 반환하면 LangGraph가 병합한다.

# chains/virtual_question_agent.py:140-179
def _read_batch(state: _VqState) -> dict:
    idx = state["batch_index"]
    batch = state["batches"][idx]

    # 컨텍스트 관리: 50장 초과면 원문 대신 요약을 누적
    if state["use_summary"]:
        context_label = "지금까지 읽은 내용 요약"
        context = "\n".join(state["context_summaries"]) if state["context_summaries"] else "(이전 구간 없음 - 첫 구간입니다)"
    else:
        context_label = "지금까지 읽은 슬라이드 원문"
        context = state["context_text"] or "(이전 구간 없음 - 첫 구간입니다)"

    prompt = _BATCH_PROMPT_TEMPLATE.format(...)

    llm = get_openai_llm()
    structured_llm = llm.with_structured_output(_BatchResult)
    result: _BatchResult = structured_llm.invoke(prompt)

    new_questions = [
        {"batch": batch["label"], "question": q.strip()}
        for q in result.questions if q.strip()
    ]

    update: dict = {                              # 바뀐 키만 반환
        "batch_index": idx + 1,                   # ← 이 증가가 루프 종료 조건이 된다
        "all_questions": state["all_questions"] + new_questions,
    }
    if state["use_summary"]:
        update["context_summaries"] = state["context_summaries"] + [f"[{batch['label']}] {result.batch_summary}"]
    else:
        update["context_text"] = (
            f"{state['context_text']}\n\n{batch['text']}" if state["context_text"] else batch["text"]
        )
    return update

여기서 LangChain 구조화 출력과 LangGraph 상태 관리가 맞물리는 지점이 _BatchResult다. 
질문 목록과 다음 구간용 요약을 한 번의 호출로 함께 받아서, 요약을 위한 별도 LLM 호출을 만들지 않는다:

def _route_after_batch(state: _VqState) -> str:

노드 2 — filter_questions (END 직전)

전 구간 질문을 모아 중복·유사 질문을 한 번 더 정리한다. 이 최종본만 화면·DB에 쓰인다.

# chains/virtual_question_agent.py:188-201
def _filter_questions(state: _VqState) -> dict:
    if not state["all_questions"]:
        return {"final_questions": []}
    listing = "\n".join(f"[{q['batch']}] {q['question']}" for q in state["all_questions"])
    prompt = _FILTER_PROMPT_TEMPLATE.format(persona_prompt=state["persona_prompt"], listing=listing)

    llm = get_openai_llm()
    structured_llm = llm.with_structured_output(_FilteredQuestionList)
    result: _FilteredQuestionList = structured_llm.invoke(prompt)
    return {"final_questions": [{"batch": q.batch, "question": q.question} for q in result.questions]}

요청마다 그래프를 새로 만들지 않는다 — 상태는 전부 invoke() 인자로만 전달되므로 컴파일된 그래프는 재사용해도 안전하다.

실행

이게 "에이전트"가 아닌 이유

파일명이 virtual_question_agent.py지만, LLM이 도구 선택이나 종료 시점을 자율 결정하지 않는다. 
그래프 위상과 반복 횟수(구간 수)는 코드가 고정하고, LLM은 각 노드에서 정해진 역할(질문 생성 / 중복 정리)만 수행한다. 
자율 판정형 tool-calling 에이전트는 이 프로젝트에서 강의자료 평가의 웹검색 한 곳뿐이다.

6. 정리 — 기능별 사용 기술

┌──────────────────┬─────────────────────────────────────────────┬───────────────────────────────┬───────────────┐
│       기능       │               LangChain 패턴                │           LangGraph           │   호출 횟수   │
├──────────────────┼─────────────────────────────────────────────┼───────────────────────────────┼───────────────┤
│ 시험문제 생성    │ with_structured_output                      │ 미사용                        │ 1회           │
├──────────────────┼─────────────────────────────────────────────┼───────────────────────────────┼───────────────┤
│ 의견형 설문 요약 │ 단발 invoke                                 │ 미사용                        │ 1회           │
├──────────────────┼─────────────────────────────────────────────┼───────────────────────────────┼───────────────┤
│ 가상질문 생성    │ with_structured_output ×2종                 │ StateGraph + 조건부 엣지 루프 │ 구간 수 + 1회 │
├──────────────────┼─────────────────────────────────────────────┼───────────────────────────────┼───────────────┤
│ 강의자료 평가    │ with_structured_output + bind_tools(웹검색) │ 미사용                        │ 1 + 최대 3회  │
└──────────────────┴─────────────────────────────────────────────┴───────────────────────────────┴───────────────┘

쓰지 않는 LangChain 기능: LCEL 파이프(|) 조합, PromptTemplate/ChatPromptTemplate 객체, output parser, memory, LangGraph checkpointer. 
프롬프트는 전부 순수 파이썬 f-string 템플릿 + .format()이고, 체인 조합 없이 llm.invoke()를 직접 부른다 — 기능당 호출이 1~2회라 추상화 계층을 더 얹을 이유가 없는 구조다.
